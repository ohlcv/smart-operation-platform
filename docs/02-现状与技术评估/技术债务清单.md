```python
# 添加 captcha 生成接口
@router.get("/auth/captcha")
def get_captcha():
    """返回验证码图片和唯一 ID"""
    import captcha
    from io import BytesIO
    
    # 生成验证码
    captcha_text, captcha_image = captcha.generate_captcha()
    captcha_id = str(uuid.uuid4())
    
    # 存储到 Redis（有效期 5 分钟）
    redis_client.setex(f"captcha:{captcha_id}", 300, captcha_text.lower())
    
    # 返回图片 base64
    buffer = BytesIO()
    captcha_image.save(buffer, format='PNG')
    img_base64 = base64.b64encode(buffer.getvalue()).decode()
    
    return {"captcha_id": captcha_id, "image": f"data:image/png;base64,{img_base64}"}
```

#### 问题 2.1.2：缺少登录锁定机制

| 属性 | 说明 |
|------|------|
| **问题描述** | 连续输错密码不限制，可无限尝试 |
| **风险等级** | 🔴 高危 |
| **影响范围** | 所有用户账号 |
| **当前状态** | 未实现 |

**解决方案：Redis 计数锁定**

```python
# backend/app/api/v1/endpoints/auth.py

LOGIN_MAX_ATTEMPTS = 5  # 最大失败次数
LOCKOUT_MINUTES = 30    # 锁定时长

def _check_login_attempts(username: str) -> bool:
    """检查是否被锁定"""
    key = f"login_attempts:{username}"
    attempts = redis_client.get(key)
    if attempts and int(attempts) >= LOGIN_MAX_ATTEMPTS:
        raise HTTPException(
            status_code=429,
            detail=f"登录失败次数过多，账号已锁定，请在 {LOCKOUT_MINUTES} 分钟后重试"
        )

def _record_failed_attempt(username: str):
    """记录失败次数"""
    key = f"login_attempts:{username}"
    pipe = redis_client.pipeline()
    pipe.incr(key)
    pipe.expire(key, LOCKOUT_MINUTES * 60)
    pipe.execute()

def _clear_attempts(username: str):
    """清除失败记录"""
    redis_client.delete(f"login_attempts:{username}")

# 在登录接口中调用
@router.post("/auth/login")
def login(...):
    _check_login_attempts(form_data.username)  # 检查是否锁定
    
    user = db.scalar(select(User).where(...))
    
    if not user or not verify_password(...):
        _record_failed_attempt(form_data.username)  # 记录失败
        raise HTTPException(status_code=401, detail="账号或密码错误")
    
    _clear_attempts(form_data.username)  # 成功后清除
    # ...
```

**预估工时**：后端 2 小时（含 Redis 配置）

```python
# backend/app/middleware/operation_log.py

from fastapi import Request
from starlette.middleware.base import BaseHTTPMiddleware
import json

class OperationLogMiddleware(BaseHTTPMiddleware):
    """操作日志中间件"""
    
    # 需要记录的操作
    SENSITIVE_METHODS = ['POST', 'PUT', 'DELETE']
    SENSITIVE_PATHS = ['/api/v1/contracts', '/api/v1/invoices', '/api/v1/customers']
    
    async def dispatch(self, request: Request, call_next):
        # 跳过非敏感请求
        if request.method not in self.SENSITIVE_METHODS:
            return await call_next(request)
        
        # 跳过非目标路径
        if not any(p in str(request.url.path) for p in self.SENSITIVE_PATHS):
            return await call_next(request)
        
        response = await call_next(request)
        
        # 异步记录日志
        if response.status_code < 400:
            await self._save_log(request, response)
        
        return response
    
    async def _save_log(self, request: Request, response):
        # 获取当前用户（需在 request.state 中存储）
        user = getattr(request.state, 'current_user', None)
        
        log = {
            'user_id': user.id if user else None,
            'username': user.username if user else 'anonymous',
            'method': request.method,
            'path': str(request.url.path),
            'body': await self._get_body(request),
            'ip': request.client.host if request.client else None,
            'user_agent': request.headers.get('user-agent'),
            'status_code': response.status_code,
            'timestamp': datetime.now().isoformat()
        }
        
        # 存储到数据库或日志文件
        await self._write_log(log)
```

```python
def mask_phone(phone: str | None) -> str:
    """脱敏手机号：138****5678"""
    if not phone or len(phone) < 7:
        return phone or ''
    return f"{phone[:3]}****{phone[-4:]}"

def mask_id_card(id_card: str | None) -> str:
    """脱敏身份证：3701***********1234"""
    if not id_card or len(id_card) < 10:
        return id_card or ''
    return f"{id_card[:4]}*********** {id_card[-4:]}"

# 在 Schema 中使用
class CustomerOut(BaseModel):
    name: str
    phone: str  # 自动脱敏
    
    @field_serializer('phone')
    def serialize_phone(self, value: str) -> str:
        return mask_phone(value)
```

```python
# backend/app/core/session.py

from redis import Redis
import json
from datetime import timedelta

class SessionManager:
    """会话管理器"""
    
    SESSION_PREFIX = "user_session:"
    SESSION_EXPIRE = 60 * 60 * 24 * 7  # 7 天
    
    def __init__(self):
        self.redis = Redis.from_url(settings.REDIS_URL)
    
    def create_session(self, user_id: int, token: str, user_info: dict):
        """创建会话"""
        key = f"{self.SESSION_PREFIX}{user_id}"
        session_data = {
            'token': token,
            'user_id': user_id,
            'username': user_info.get('username'),
            'login_time': datetime.now().isoformat(),
            'ip': user_info.get('ip'),
            'user_agent': user_info.get('user_agent')
        }
        self.redis.setex(key, self.SESSION_EXPIRE, json.dumps(session_data))
    
    def get_online_users(self) -> list:
        """获取所有在线用户"""
        keys = self.redis.keys(f"{self.SESSION_PREFIX}*")
        users = []
        for key in keys:
            data = self.redis.get(key)
            if data:
                users.append(json.loads(data))
        return users
    
    def force_logout(self, user_id: int):
        """强制下线"""
        self.redis.delete(f"{self.SESSION_PREFIX}{user_id}")
```
